- Deviated Cases Dynamic: 
VAR SelectedType =
    SELECTEDVALUE(
        'Delay Type Slicer'[Delay Type],
        "Accept Time"
    )

VAR ThresholdSeconds =
    SELECTEDVALUE(
        'Accept Time Threshold'[Threshold Minutes],
        2
    ) * 60

RETURN
    SWITCH(
        SelectedType,
        "Accept Time",
            CALCULATE(
                DISTINCTCOUNT(ERR[Event Id]),
                ERR[Ambulance Assigned] = 1,
                FILTER(
                    ERR,
                    NOT ISBLANK(ERR[Assign Time])
                        && NOT ISBLANK(ERR[Accepted Time])
                        && DATEDIFF(
                            ERR[Assign Time],
                            ERR[Accepted Time],
                            SECOND
                        ) > ThresholdSeconds
                )
            ),
        "On Action Time",
            CALCULATE(
                DISTINCTCOUNT(ERR[Event Id]),
                ERR[Ambulance Assigned] = 1,
                FILTER(
                    ERR,
                    NOT ISBLANK(ERR[Accepted Time])
                        && NOT ISBLANK(ERR[On Action Time])
                        && DATEDIFF(
                            ERR[Accepted Time],
                            ERR[On Action Time],
                            SECOND
                        ) > ThresholdSeconds
                )
            )
    )
